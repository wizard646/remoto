// los iconos se dibujan con lineas: no dependen de ninguna fuente
// de iconos de Windows, asi que se ven igual en cualquier equipo
static class Iconos {
    static GraphicsPath Redondeado(int x, int y, int w, int h, int r) {
        if (r > w / 2) r = w / 2;
        if (r > h / 2) r = h / 2;
        var p = new GraphicsPath();
        p.AddArc(x, y, r, r, 180, 90);
        p.AddArc(x + w - r, y, r, r, 270, 90);
        p.AddArc(x + w - r, y + h - r, r, r, 0, 90);
        p.AddArc(x, y + h - r, r, r, 90, 90);
        p.CloseFigure();
        return p;
    }

    public static void Dibuja(Graphics g, string cual, int cx, int cy, int s, Color c, Color fondo) {
        g.SmoothingMode = SmoothingMode.AntiAlias;
        float l = Math.Max(1.6f, s * 0.085f);
        using (var p = new Pen(c, l) { StartCap = LineCap.Round, EndCap = LineCap.Round, LineJoin = LineJoin.Round })
        using (var b = new SolidBrush(c)) {
            float m = s * 0.5f;
            int q = (int)(s * 0.18f);
            switch (cual) {
                case "play":
                    g.FillPolygon(b, new Point[] {
                        new Point((int)(cx - m * 0.55f), (int)(cy - m)),
                        new Point((int)(cx - m * 0.55f), (int)(cy + m)),
                        new Point((int)(cx + m), (int)cy) });
                    break;
                case "pausa":
                    g.FillRectangle(b, cx - (int)(m * 0.7f), cy - (int)m, q, (int)(m * 2));
                    g.FillRectangle(b, cx + (int)(m * 0.15f), cy - (int)m, q, (int)(m * 2));
                    break;
                case "nota":
                    g.FillEllipse(b, cx - (int)(m * 1.1f), cy + (int)(m * 0.3f), q + (int)(m * 0.5f), q + (int)(m * 0.5f));
                    g.DrawLine(p, cx + (int)(m * 0.4f), cy + (int)(m * 0.55f), cx + (int)(m * 0.4f), cy - (int)(m * 0.9f));
                    g.DrawLine(p, cx + (int)(m * 0.4f), cy - (int)(m * 0.9f), cx + (int)(m * 1.1f), cy - (int)(m * 0.5f));
                    break;
                case "papel":
                    using (var pa = Redondeado(cx - (int)m, cy - (int)m, (int)(m * 2), (int)(m * 2), q))
                        g.DrawPath(p, pa);
                    g.DrawLine(p, cx - (int)(m * 0.5f), cy - (int)(m * 0.35f), cx + (int)(m * 0.5f), cy - (int)(m * 0.35f));
                    g.DrawLine(p, cx - (int)(m * 0.5f), cy + (int)(m * 0.05f), cx + (int)(m * 0.5f), cy + (int)(m * 0.05f));
                    g.DrawLine(p, cx - (int)(m * 0.5f), cy + (int)(m * 0.45f), cx + (int)(m * 0.2f), cy + (int)(m * 0.45f));
                    break;
                case "carpeta":
                    g.DrawLine(p, cx - (int)m, cy - (int)(m * 0.6f), cx - (int)(m * 0.2f), cy - (int)(m * 0.6f));
                    g.DrawLine(p, cx - (int)(m * 0.2f), cy - (int)(m * 0.6f), cx - (int)(m * 0.05f), cy - (int)(m * 0.25f));
                    g.DrawLine(p, cx - (int)(m * 0.05f), cy - (int)(m * 0.25f), cx + (int)m, cy - (int)(m * 0.25f));
                    g.DrawLine(p, cx - (int)m, cy - (int)(m * 0.6f), cx - (int)m, cy + (int)m);
                    g.DrawLine(p, cx + (int)m, cy - (int)(m * 0.25f), cx + (int)m, cy + (int)m);
                    g.DrawLine(p, cx - (int)m, cy + (int)m, cx + (int)m, cy + (int)m);
                    break;
                case "codigo":
                    g.DrawLine(p, cx - (int)(m * 0.55f), cy - (int)m, cx - (int)m, cy);
                    g.DrawLine(p, cx - (int)m, cy, cx - (int)(m * 0.55f), cy + (int)m);
                    g.DrawLine(p, cx + (int)(m * 0.55f), cy - (int)m, cx + (int)m, cy);
                    g.DrawLine(p, cx + (int)m, cy, cx + (int)(m * 0.55f), cy + (int)m);
                    g.DrawLine(p, cx + (int)(m * 0.15f), cy - (int)(m * 0.9f), cx - (int)(m * 0.15f), cy + (int)(m * 0.9f));
                    break;
                case "luna":
                    g.FillEllipse(b, cx - (int)m, cy - (int)m, (int)(m * 2), (int)(m * 2));
                    using (var b2 = new SolidBrush(fondo))
                        g.FillEllipse(b2, cx - (int)(m * 0.1f), cy - (int)(m * 1.2f), (int)(m * 2), (int)(m * 2));
                    break;
                case "candado":
                    g.DrawArc(p, cx - (int)(m * 0.6f), cy - (int)m, (int)(m * 1.2f), (int)(m * 1.2f), 180, 180);
                    using (var pa = Redondeado(cx - (int)m, cy - (int)(m * 0.2f), (int)(m * 2), (int)(m * 1.2f), q / 2))
                        g.DrawPath(p, pa);
                    break;
                case "encender":
                    g.DrawArc(p, cx - (int)(m * 0.9f), cy - (int)(m * 0.9f), (int)(m * 1.8f), (int)(m * 1.8f), -60, 300);
                    g.DrawLine(p, cx, cy - (int)(m * 1.15f), cx, cy - (int)(m * 0.35f));
                    break;
                case "reiniciar":
                    g.DrawArc(p, cx - (int)(m * 0.9f), cy - (int)(m * 0.9f), (int)(m * 1.8f), (int)(m * 1.8f), 40, 280);
                    g.FillPolygon(b, new Point[] {
                        new Point(cx + (int)(m * 0.5f), cy - (int)(m * 1.15f)),
                        new Point(cx + (int)(m * 1.1f), cy - (int)(m * 0.5f)),
                        new Point(cx + (int)(m * 0.25f), cy - (int)(m * 0.4f)) });
                    break;
                case "salir":
                    g.DrawLine(p, cx - (int)(m * 0.7f), cy - (int)(m * 0.7f), cx + (int)(m * 0.7f), cy + (int)(m * 0.7f));
                    g.DrawLine(p, cx + (int)(m * 0.7f), cy - (int)(m * 0.7f), cx - (int)(m * 0.7f), cy + (int)(m * 0.7f));
                    break;
            }
        }
    }
}

class Boton : Control {
    public string Ico = "", Rotulo = "";
    Color tinte = Color.FromArgb(245, 245, 247);
    bool encima = false;

    public Boton() {
        SetStyle(ControlStyles.AllPaintingInWmPaint | ControlStyles.OptimizedDoubleBuffer |
                 ControlStyles.UserPaint | ControlStyles.ResizeRedraw, true);
        BackColor = Color.FromArgb(24, 24, 28);
        Cursor = Cursors.Hand;
        Font = new Font("Segoe UI", 8.5f);
        Size = new Size(112, 40);
    }

    protected override void OnPaint(PaintEventArgs e) {
        var g = e.Graphics;
        g.Clear(Parent != null ? Parent.BackColor : BackColor);
        var r = new Rectangle(1, 1, Width - 2, Height - 2);
        int rad = 11;
        var ruta = new GraphicsPath();
        ruta.AddArc(r.X, r.Y, rad, rad, 180, 90);
        ruta.AddArc(r.Right - rad, r.Y, rad, rad, 270, 90);
        ruta.AddArc(r.Right - rad, r.Bottom - rad, rad, rad, 0, 90);
        ruta.AddArc(r.X, r.Bottom - rad, rad, rad, 90, 90);
        ruta.CloseFigure();

        bool luz = Enabled && encima;
        using (var b = new SolidBrush(luz ? Color.FromArgb(34, 34, 40) : Color.FromArgb(24, 24, 28)))
            g.FillPath(b, ruta);
        using (var p = new Pen(luz ? Color.FromArgb(58, 58, 66) : Color.FromArgb(32, 32, 37)))
            g.DrawPath(p, ruta);

        Color c = Enabled ? tinte : Color.FromArgb(80, 80, 88);
        var t = new StringFormat { Alignment = StringAlignment.Center, LineAlignment = StringAlignment.Center };
        using (var b = new SolidBrush(c)) {
            SizeF ts = g.MeasureString(Rotulo, Font);
            if (ts.Width > Width - 40) {
                // rotulo largo: icono solo y el texto al lado, chico
                Iconos.Dibuja(g, Ico, 26, Height / 2, 22, c, BackColor);
                g.DrawString(Rotulo, Font, b, new RectangleF(42, 0, Width - 46, Height), t);
            } else {
                Iconos.Dibuja(g, Ico, 22, Height / 2, 22, c, BackColor);
                g.DrawString(Rotulo, Font, b, new RectangleF(40, 0, Width - 44, Height), t);
            }
        }
    }

    protected override void OnMouseEnter(EventArgs e) { encima = true; Invalidate(); base.OnMouseEnter(e); }
    protected override void OnMouseLeave(EventArgs e) { encima = false; Invalidate(); base.OnMouseLeave(e); }
    protected override void OnEnabledChanged(EventArgs e) { Invalidate(); base.OnEnabledChanged(e); }
}

    // ---------------- el logo, embebido en el exe --------------------
    static Icon Logo(int lado) {
        try {
            using (var s = Assembly.GetExecutingAssembly().GetManifestResourceStream("Remoto.ico")) {
                if (s != null) return new Icon(s, lado, lado);
            }
        } catch { }

    string RutaExe() { return "\"" + Assembly.GetExecutingAssembly().Location + "\""; }

    bool LeoInicio() {
        try {
            using (var k = Registry.CurrentUser.OpenSubKey(CLAVE_INICIO))
                if (k != null) return (k.GetValue("Remoto") as string) == RutaExe();
        } catch { }
        return false;
    }

    bool EscriboInicio(bool activar) {
        try {
            using (var k = Registry.CurrentUser.CreateSubKey(CLAVE_INICIO)) {
                if (activar) k.SetValue("Remoto", RutaExe());
                else k.DeleteValue("Remoto", false);
            }
            return true;
        } catch { return false; }
    }

    // ---------------- refresco ---------------------------------------
    void Refrescar() {
        lblNombre.Text = "Equipo: " + Environment.MachineName;
        lblIp.Text = "http://" + IpLocal() + ":8080";
        try { lblClave.Text = "clave  " + File.ReadAllText(CLAVE).Trim(); } catch { lblClave.Text = "clave  ?"; }

        bool vivo = Vivo();
        if (estabaVivo && !vivo && !cerrando) { Arrancar(); vivo = Vivo(); }
        estabaVivo = vivo;

        btnPrincipal.Rotulo = vivo ? "Detener" : "Iniciar";
        btnPrincipal.Ico = vivo ? "salir" : "encender";
        btnPrincipal.Invalidate();
        lblEstado.Text = vivo ? "Encendido. Abre la app Remoto en tu celular." : "Detenido. Se esta levantando...";
        lblEstado.ForeColor = vivo ? Tenue : Color.FromArgb(255, 90, 82);

        if ((DateTime.Now - ultimoCelular).TotalSeconds > 3) { ultimoCelular = DateTime.Now; Cliente(); }
    }

    void Cliente() {
        string t = "Celular:  sin conectar";
        var j = Pide("/api/clientes");
        if (j != null) {
            var l = MiniJson.Clientes(j);
            if (l != null && l.Count > 0) {
                // gana el que se movio hace menos tiempo
                Dictionary<string, string> p = l[0];
                foreach (var d in l) {
                    int a = 999999, b = 999999;
                    int.TryParse(d.ContainsKey("hace") ? d["hace"] : "", out a);
                    int.TryParse(p.ContainsKey("hace") ? p["hace"] : "", out b);
                    if (a < b) p = d;
                }
                string nom = p.ContainsKey("nombre") ? p["nombre"] : "";
                string dir = p.ContainsKey("ip") ? p["ip"] : "";
                if (string.IsNullOrWhiteSpace(nom)) nom = "Celular";
                bool viva = p.ContainsKey("activa") && p["activa"] == "true";
                t = "Celular:  " + nom + "   " + dir;
                if (!viva) t = "Celular:  sin actividad";
            }
        }
        if (lblCelular.Text != t) lblCelular.Text = t;
    }

    // ---------------- red y proceso -----------------------------------
    string IpLocal() {
        List<string> otras = new List<string>();
        try {
            foreach (var ni in NetworkInterface.GetAllNetworkInterfaces()) {
                if (ni.OperationalStatus != OperationalStatus.Up) continue;
                if (ni.NetworkInterfaceType == NetworkInterfaceType.Loopback) continue;
                if (ni.NetworkInterfaceType == NetworkInterfaceType.Tunnel) continue;
                string nom = ni.Name.ToLower();
                if (nom.Contains("virtual") || nom.Contains("vmware") || nom.Contains("bluetooth")) continue;
                bool sale = ni.GetIPProperties().GatewayAddresses.Count > 0;
                foreach (var ua in ni.GetIPProperties().UnicastAddresses) {
                    if (ua.Address.AddressFamily != AddressFamily.InterNetwork) continue;
                    if (IPAddress.IsLoopback(ua.Address)) continue;
                    if (ua.Address.ToString().StartsWith("169.254.")) continue;
                    if (sale) return ua.Address.ToString();
                    otras.Add(ua.Address.ToString());
                }
            }
        } catch { }
        return otras.Count > 0 ? otras[0] : "?";
    }

    bool Vivo() {
        try {
            var c = new TcpClient();
            var ar = c.BeginConnect("127.0.0.1", 8080, null, null);
            bool ok = ar.AsyncWaitHandle.WaitOne(400, false) && c.Connected;
            c.Close();
            return ok;
        } catch { return false; }
    }

    void Arrancar() {
        try {
            var psi = new ProcessStartInfo("powershell.exe",
                "-NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -File \"" + SERVIDOR + "\"") {
                CreateNoWindow = true, UseShellExecute = false,
                WindowStyle = ProcessWindowStyle.Hidden, WorkingDirectory = RAIZ
            };
            servidorProc = Process.Start(psi);
        } catch { }
    }

    void Detener() {
        try { if (servidorProc != null && !servidorProc.HasExited) { servidorProc.Kill(); servidorProc = null; } } catch { }
        try {
            var psi = new ProcessStartInfo("cmd.exe",
                "/c for /f \"tokens=5\" %a in ('netstat -ano ^| findstr :8080 ^| findstr LISTENING') do taskkill /PID %a /F") {
                CreateNoWindow = true, UseShellExecute = false, WindowStyle = ProcessWindowStyle.Hidden
            };
            Process.Start(psi).WaitForExit(6000);
        } catch { }
        estabaVivo = false;
    }

    // ---------------- pedirle cosas al servidor -----------------------
    string Pide(string ruta) {
        string clave = "";
        try { clave = File.ReadAllText(CLAVE).Trim(); } catch { }
        try {
            var req = (HttpWebRequest)WebRequest.Create("http://127.0.0.1:8080" + ruta);
            req.Method = "GET";
            req.Timeout = 6000;
            req.ReadWriteTimeout = 6000;
            req.Headers.Add("x-token", clave);
            using (var r = (HttpWebResponse)req.GetResponse())
            using (var sr = new StreamReader(r.GetResponseStream())) return sr.ReadToEnd();
        } catch (WebException) {
            // el servidor esta ocupado con otra peticion. no es un error:
            // se reintenta en el siguiente refresh y no se anota nada
            return null;
        } catch (Exception ex) {
            try { File.AppendAllText(Path.Combine(RAIZ, "error.log"),
                     DateTime.Now.ToString("HH:mm:ss") + "  " + ex.Message + "\r\n"); } catch { }
            return null;
        }
    }

    void CerrarTodo() {
        timer.Stop();
        try { if (servidorProc != null && !servidorProc.HasExited) servidorProc.Kill(); } catch { }
        icono.Visible = false;
        Application.Exit();
    }

    protected override void OnFormClosing(FormClosingEventArgs e) {
        if (!cerrando && e.CloseReason == CloseReason.UserClosing) {
            e.Cancel = true;
            Hide();
            icono.BalloonTipTitle = "Remoto";
            icono.BalloonTipText = "Sigue funcionando en segundo plano.";
            icono.ShowBalloonTip(2500);
            return;
        }
        CerrarTodo();
    }
